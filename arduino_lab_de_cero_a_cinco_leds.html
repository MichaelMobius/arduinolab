1. Bucle while + barrido con += y Serial.println() 
void setup() {
  Serial.begin(9600);
  int p = 8;
  while (p <= 12) {
    pinMode(p, OUTPUT);
    p += 1;
  }
}

void loop() {
  int pin = 8;
  while (pin <= 12) {
    Serial.print("Encendiendo pin D");
    Serial.println(pin);
    digitalWrite(pin, HIGH);
    delay(250);
    digitalWrite(pin, LOW);
    pin += 1;
  }
}
2. Espera activa con while (digitalRead(boton) == HIGH) 
const int led = 8;
const int boton = 2;
int contador = 0;

void setup() {
  Serial.begin(9600);
  pinMode(led, OUTPUT);
  pinMode(boton, INPUT_PULLUP);
  Serial.println("Sistema listo. Pulsa el boton fisico SW1...");
}

void loop() {
  // Espera mientras el boton este suelto (HIGH)
  while (digitalRead(boton) == HIGH) {
    digitalWrite(led, LOW);
  }
  contador += 1;
  Serial.print("Pulsacion detectada #");
  Serial.println(contador);
  digitalWrite(led, HIGH);
  delay(300);
}
